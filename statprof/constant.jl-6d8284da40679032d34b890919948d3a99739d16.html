123 """
124 ConstantInt(typ::LLVM.IntegerType, val, [signed=false])
125
126 Create a constant integer value of the given type and value. If `signed` is `true`, the
127 value is treated as a signed integer.
128 """
129 function ConstantInt(typ::IntegerType, val::Integer, signed=false)
130 valbits = ceil(Int, log2(abs(val))) + 1 # FIXME: doesn't work for val=0
131 numwords = ceil(Int, valbits / 64)
132 words = Vector{Culonglong}(undef, numwords)
133 for i in 1:numwords
134 words[i] = (val >> 64(i-1)) % Culonglong
135 end
136 return ConstantInt(API.LLVMConstIntOfArbitraryPrecision(typ, numwords, words))
137 end

216 abstract type ConstantDataSequential <: Constant end
217
218 # ConstantData can only contain primitive types (1/2/4/8 byte integers, float/half),
219 # as opposed to ConstantAggregate which can contain arbitrary LLVM values.
220 #
221 # however, LLVM seems to use both array types interchangeably, e.g., constructing
222 # a ConstArray through LLVMConstArray may return a ConstantDataArray (presumably as an
223 # optimization, when the data can be represented as densely packed primitive values).
224 # because of that, ConstantDataArray and ConstantArray need to behave the same way,
225 # concretely, indexing a ConstantDataArray has to return LLVM constant values...
226 #
227 # XXX: maybe we should just not expose ConstantDataArray then?
228 # one advantage of keeping them separate is that creating a ConstantDataArray
229 # is much cheaper (we should also be able to iterate much more efficiently,
230 # but cannot support that as explained above).
231

234 Base.length(cda::ConstantDataSequential) = length(value_type(cda))

236 function Base.getindex(cda::ConstantDataSequential, idx::Integer)
237 @boundscheck 1 <= idx <= length(cda) || throw(BoundsError(cda, idx))
238 Value(API.LLVMGetElementAsConstant(cda, idx-1))
239 end

241 constants = Array{Value}(undef, length(cda))

412 function Base.size(ca::ConstantArray)
413 dims = Int[]
414 typ = value_type(ca)
415 while typ isa ArrayType
416 push!(dims, length(typ))
417 typ = eltype(typ)
418 end

421 Base.length(ca::ConstantArray) = prod(size(ca))

423
424 function Base.getindex(ca::ConstantArray, idx::Integer...)
425 # multidimensional arrays are represented by arrays of arrays,
426 # which we need to 'peel back' by looking at the operand sets.
427 # for the final dimension, we use LLVMGetElementAsConstant
428 @boundscheck Base.checkbounds_indices(Bool, axes(ca), idx) ||
429 throw(BoundsError(ca, idx))
430 I = CartesianIndices(size(ca))[idx...]
431 for i in Tuple(I)
432 if isempty(operands(ca))
433 # XXX: is this valid? LLVMGetElementAsConstant is meant to be used with
434 # Constant*Data*Arrays, not ConstantArrays
435 ca = Value(API.LLVMGetElementAsConstant(ca, i-1))
436 else
437 ca = (Base.@_propagate_inbounds_meta; operands(ca)[i])
438 end
439 end
440 return ca
441 end

476 """
477 ConstantStruct(value::T, [name=String(nameof(T)), anonymous=false, packed=false])
478
479 Create a constant struct from an (isbits) Julia struct instance.
480 """
481 function ConstantStruct(value::T, name::AbstractString=String(nameof(T));
482 anonymous::Bool=false, packed::Bool=false) where {T}
483 isbitstype(T) || throw(ArgumentError("Can only create a ConstantStruct from an isbits struct"))
484 isprimitivetype(T) && throw(ArgumentError("Cannot create a ConstantStruct from a primitive value"))
485
486 constants = Vector{Constant}()
487 for fieldname in fieldnames(T)
488 field = getfield(value, fieldname)
489
490 if isa(field, Integer)
491 push!(constants, ConstantInt(field))
492 elseif isa(field, AbstractFloat)
493 push!(constants, ConstantFP(field))
494 else # TODO: nested structs?
495 throw(ArgumentError("only structs with boolean, integer and floating point fields are allowed"))
496 end
497 end
498
499 if anonymous
500 ConstantStruct(constants; packed)
501 elseif haskey(types(context()), name)
502 typ = types(context())[name]
503 if collect(elements(typ)) != value_type.(constants)
504 throw(ArgumentError("Cannot create struct $name {$(join(value_type.(constants), ", "))} as it is already defined in this context as {$(join(elements(typ), ", "))}."))
505 end
506 ConstantStruct(typ, constants)
507 else
508 typ = StructType(name)
509 elements!(typ, value_type.(constants))
510 ConstantStruct(typ, constants)
511 end
512 end
