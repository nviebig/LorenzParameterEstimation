147 function replace_metadata_uses!(old::Value, new::Value)
148 if value_type(old) == value_type(new)
149 API.LLVMReplaceAllMetadataUsesWith(old, new)
150 else
151 # NOTE: LLVM does not support replacing values of different types, either using
152 # regular RAUW or only on metadata. The latter should probably be supported.
153 # Instead, we replace by a bitcast to the old type.
154 compat_new = const_bitcast(new, value_type(old))
155 replace_metadata_uses!(old, compat_new)
156
157 # the above is often invalid, e.g. for module-level metadata identifying functions.
158 # so we peek into such metadata and try to get rid of the bitcast. see also
159 # https://discourse.llvm.org/t/replacing-module-metadata-uses-of-function/62431/4
160 mod = LLVM.parent(new)
161 while !isa(mod, LLVM.Module)
162 mod = LLVM.parent(new)
163 end
164 function recurse(md)

166 if op isa ValueAsMetadata && Value(op) == compat_new
167 LLVM.replace_operand(md, i, Metadata(new))
168 elseif isa(op, MDTuple)
169 recurse(op)
170 end
171 end
172 end
173 for (key, md) in metadata(mod)
