129 """
130 NewPMPassBuilder(; verify_each=false, debug_logging=false, pipeline_tuning_kwargs...)
131
132 Create a new pass builder. The pass builder is the main object used to construct and run
133 pass pipelines. The `verify_each` keyword argument enables module verification after each
134 pass, while `debug_logging` can be used to enable more output. Pass builder objects needs to
135 be disposed after use.
136
137 Several other keyword arguments can be used to tune the pipeline. This only has an effect
138 when using one of LLVM's default pipelines, like `default<O3>`:
139
140 - `loop_interleaving::Bool=false`: Enable loop interleaving.
141 - `loop_vectorization::Bool=false`: Enable loop vectorization.
142 - `slp_vectorization::Bool=false`: Enable SLP vectorization.
143 - `loop_unrolling::Bool=false`: Enable loop unrolling.
144 - `forget_all_scev_in_loop_unroll::Bool=false`: Forget all SCEV information in loop
145 unrolling.
146 - `licm_mssa_opt_cap::Int=0`: LICM MSSA optimization cap.
147 - `licm_mssa_no_acc_for_promotion_cap::Int=0`: LICM MSSA no access for promotion cap.
148 - `call_graph_profile::Bool=false`: Enable call graph profiling.
149 - `merge_functions::Bool=false`: Enable function merging.
150
151 After a pass builder is constructed, custom passes can be registered with `register!`,
152 passes or nested pass managers can be added with `add!`, and finally the passes can be run
153 with `run!`:
154
155 ```julia
156 @dispose pb = NewPMPassBuilder(verify_each=true) begin
157 register!(pb, SomeCustomPass())
158 add!(pb, SomeModulePass())
159 add!(pb, NewPMFunctionPassManager()) do fpm
160 add!(fpm, SomeFunctionPass())
161 end
162 run!(pb, mod, tm)
163 end
164 ```
165
166 For quickly running a simple pass or pipeline, a shorthand `run!` method is provided that
167 obviates the construction of a `NewPMPassBuilder`:
168
169 ```julia
170 run!("some-pass", mod, tm; verify_each=true)
171 ```
172
173 See also: [`register!`](@ref), [`add!`](@ref), [`run!`](@ref)
174 """

361 function define_pass(pass_name, class_name, define_class=true)
362 # don't re-define passes (some work with multiple types of managers,
363 # or could be manually-defined)
364 if isdefined(LLVM, class_name)
365 return
366 end
367
368 ex = quote
369 export $(esc(class_name))
370 end
371 if define_class
372 push!(ex.args, :(
373 function $(esc(class_name))(; kwargs...)
374 return $pass_name * kwargs_to_params(kwargs)
375 end
376 ))
377 end
378 return ex
379 end
