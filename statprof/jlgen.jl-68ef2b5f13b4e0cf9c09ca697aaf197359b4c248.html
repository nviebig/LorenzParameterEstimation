46 """
47 methodinstance(ft::Type, tt::Type, [world::UInt])
48
49 Look up the method instance that corresponds to invoking the function with type `ft` with
50 argument typed `tt`. If the `world` argument is specified, the look-up is static and will
51 always return the same result. If the `world` argument is not specified, the look-up is
52 dynamic and the returned method instance will depende on the current world age. If no method
53 is found, a `MethodError` is thrown.
54
55 This function is highly optimized, and results do not need to be cached additionally.
56
57 Only use this function with concrete signatures, i.e., using the types of values you would
58 pass at run time. For non-concrete signatures, use `generic_methodinstance` instead.
59
60 """

104 function methodinstance_generator(world::UInt, source, self, ft::Type, tt::Type)
105 @nospecialize
106 @assert CC.isType(ft) && CC.isType(tt)
107 ft = ft.parameters[1]
108 tt = tt.parameters[1]
109
110 stub = Core.GeneratedFunctionStub(identity, Core.svec(:methodinstance, :ft, :tt), Core.svec())
111
112 # look up the method match
113 method_error = :(throw(MethodError(ft, tt, $world)))
114 sig = Tuple{ft, tt.parameters...}
115 min_world = Ref{UInt}(typemin(UInt))
116 max_world = Ref{UInt}(typemax(UInt))
117 match = ccall(:jl_gf_invoke_lookup_worlds, Any,
118 (Any, Any, Csize_t, Ref{Csize_t}, Ref{Csize_t}),
119 sig, #=mt=# nothing, world, min_world, max_world)
120 match === nothing && return stub(world, source, method_error)
121
122 # look up the method and code instance
123 mi = ccall(:jl_specializations_get_linfo, Ref{MethodInstance},
124 (Any, Any, Any), match.method, match.spec_types, match.sparams)
125 ci = CC.retrieve_code_info(mi, world)
126
127 # prepare a new code info
128 new_ci = copy(ci)
129 empty!(new_ci.code)
130 empty!(new_ci.codelocs)
131 empty!(new_ci.linetable)
132 empty!(new_ci.ssaflags)
133 new_ci.ssavaluetypes = 0
134
135 # propagate edge metadata
136 new_ci.min_world = min_world[]
137 new_ci.max_world = max_world[]
138 new_ci.edges = MethodInstance[mi]
139
140 # prepare the slots
141 new_ci.slotnames = Symbol[Symbol("#self#"), :ft, :tt]
142 new_ci.slotflags = UInt8[0x00 for i = 1:3]
143
144 # return the method instance
145 push!(new_ci.code, CC.ReturnNode(mi))
146 push!(new_ci.ssaflags, 0x00)
147 push!(new_ci.linetable, @LineInfoNode(methodinstance))
148 push!(new_ci.codelocs, 1)
149 new_ci.ssavaluetypes += 1
150
151 return new_ci
152 end
