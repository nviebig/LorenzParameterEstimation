93 # helper to build a PdfConfig from nbins/halfwidth (or explicit centers)
94 function make_pdf_config(::Type{T};
95 nbins::Int=16,
96 range_halfwidth::Real=30.0,
97 centers::Union{Nothing,AbstractVector}=nothing,
98 bandwidth::Union{Nothing,Real}=nothing,
99 loss_mode::Symbol=:kl,
100 sinkhorn_ε::Real=1e-2,
101 sinkhorn_iters::Int=50) where {T}
102
103 C = centers === nothing ?
104 collect(range(T(-range_halfwidth), T(range_halfwidth); length=nbins)) :
105 T.(centers)
106
107 Δ = length(C) >= 2 ? (C[end]-C[1])/(length(C)-1) : T(1)
108 h = bandwidth === nothing ? T(0.5*Δ) : T(bandwidth)
109
110 return PdfConfig{T}(C, h, loss_mode, T(sinkhorn_ε), sinkhorn_iters)
111 end
