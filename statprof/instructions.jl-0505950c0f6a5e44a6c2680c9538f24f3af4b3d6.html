652 - `arcp`: allow use of reciprocal rather than perform division
653 - `contract`: allow contraction of operations
654 - `afn`: allow substitution of approximate calculations for functions
655 - `reassoc`: allow reassociation of operations

671 afn && (flags |= LLVM.API.LLVMFastMathApproxFunc)

673 API.LLVMSetFastMathFlags(inst, flags)
674 end
