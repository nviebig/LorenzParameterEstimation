708 timeout given by `timeout_s` seconds.
709
710 The keyword arguments determine which of read and/or write status should be monitored; at
711 least one of them must be set to `true`.
712
713 The returned value is an object with boolean fields `readable`, `writable`, and `timedout`,
714 giving the result of the polling.
715 """
716 function poll_fd(s::Union{RawFD, Sys.iswindows() ? WindowsRawSocket : Union{}}, timeout_s::Real=-1; readable=false, writable=false)
717 mask = FDEvent(readable, writable, false, false)
718 mask.timedout && return mask
719 fdw = _FDWatcher(s, mask)
720 local timer
721 # we need this flag to explicitly track whether we call `close` already, to update the internal refcount correctly
722 timedout = false # TODO: make this atomic
723 try
724 if timeout_s >= 0
725 # delay creating the timer until shortly before we start the poll wait
726 timer = Timer(timeout_s) do t
727 timedout && return
728 timedout = true
729 close(fdw, mask)
730 end
731 try
732 while true
733 events = _wait(fdw, mask)
734 if timedout || !events.timedout
735 @lock fdw.notify fdw.events &= ~events.events
736 return events
737 end
738 end
739 catch ex
740 ex isa EOFError || rethrow()
741 return FDEvent()
742 end
743 else
744 return wait(fdw, mask)
745 end
746 finally
747 if @isdefined(timer)
748 if !timedout
749 timedout = true
750 close(timer)
751 close(fdw, mask)
752 end
753 else
754 close(fdw, mask)
755 end
756 end

791 watch_folder(path::AbstractString, timeout_s::Real=-1)
792
793 Watches a file or directory `path` for changes until a change has occurred or `timeout_s`
794 seconds have elapsed. This function does not poll the file system and instead uses platform-specific

797

799 `unwatch_folder` is called on the same `path`.
800

803 giving the event.
804

806 <https://nodejs.org/api/fs.html#fs_caveats> for more detailed information.

809 function watch_folder(s::String, timeout_s::Real=-1)
810 fm = get!(watched_folders, s) do
811 return FolderMonitor(s)
812 end
813 local timer
814 if timeout_s >= 0
815 @lock fm.notify isempty(fm.channel) || return popfirst!(fm.channel)
816 if timeout_s <= 0.010
817 # for very small timeouts, we can just sleep for the whole timeout-interval
818 (timeout_s == 0) ? yield() : sleep(timeout_s)
819 @lock fm.notify isempty(fm.channel) || return popfirst!(fm.channel)
820 return "" => FileEvent() # timeout
821 else
822 timer = Timer(timeout_s) do t
823 @lock fm.notify notify(fm.notify)
824 end
825 end
826 end
827 # inline a copy of `wait` with added support for checking timer
828 fm.handle == C_NULL && throw(EOFError())
829 preserve_handle(fm)
830 lock(fm.notify)
831 evt = try
832 fm.handle == C_NULL && throw(EOFError())
833 while isempty(fm.channel)
834 if @isdefined(timer)
835 isopen(timer) || return "" => FileEvent() # timeout
836 end
837 wait(fm.notify)
838 end
839 popfirst!(fm.channel)
840 finally
841 unlock(fm.notify)
842 unpreserve_handle(fm)
843 @isdefined(timer) && close(timer)
844 end

864 """
865 poll_file(path::AbstractString, interval_s::Real=5.007, timeout_s::Real=-1) -> (previous::StatStruct, current)
866
867 Monitor a file for changes by polling every `interval_s` seconds until a change occurs or
868 `timeout_s` seconds have elapsed. The `interval_s` should be a long period; the default is
869 5.007 seconds.
870
871 Returns a pair of status objects `(previous, current)` when a change is detected.
872 The `previous` status is always a `StatStruct`, but it may have all of the fields zeroed
873 (indicating the file didn't previously exist, or wasn't previously accessible).
874
875 The `current` status object may be a `StatStruct`, an `EOFError` (indicating the timeout elapsed),
876 or some other `Exception` subtype (if the `stat` operation failed - for example, if the path does not exist).
877
878 To determine when a file was modified, compare `current isa StatStruct && mtime(prev) != mtime(current)` to detect
879 notification of changes. However, using [`watch_file`](@ref) for this operation is preferred, since
880 it is more reliable and efficient, although in some situations it may not be available.
881 """
882 function poll_file(s::AbstractString, interval_seconds::Real=5.007, timeout_s::Real=-1)
883 pfw = PollingFileWatcher(s, Float64(interval_seconds))
884 local timer
885 try
886 if timeout_s >= 0
887 timer = Timer(timeout_s) do t
888 close(pfw)
889 end
890 end
891 statdiff = wait(pfw)
892 if isa(statdiff[2], IOError)
893 # file didn't initially exist, continue watching for it to be created (or the error to change)
894 statdiff = wait(pfw)
895 end
896 return statdiff
897 finally
898 close(pfw)
899 @isdefined(timer) && close(timer)
900 end
901 end
