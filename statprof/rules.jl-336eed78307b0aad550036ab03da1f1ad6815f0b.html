132 function apply!(o::RMSProp, state, x::AbstractArray{T}, dx) where T
133 η, ρ, ϵ = T(o.eta), T(o.rho), _eps(T, o.epsilon)
134 quad, lin = state

179 function apply!(o::Rprop, state, x::AbstractArray{T}, dx) where T
180 ℓ, Γ = T.(o.ell), T.(o.gamma)
181 g, η = state
182
183 η = broadcast(g, η, dx) do g, η, dx
184 g * dx > 0 ? min(η * ℓ[2], Γ[2]) : g * dx < 0 ? max(η * ℓ[1], Γ[1]) : η
185 end
186 g = broadcast(g, dx) do g, dx

188 end

190
191 return (g, η), dx′
192 end

280 η, β, ϵ = T(o.eta), T.(o.beta), _eps(T, o.epsilon)
281 ρ∞ = 2/(1-β[2]) - 1 |> real
282

284
285 @.. mt = β[1] * mt + (1 - β[1]) * dx
286 @.. vt = β[2] * vt + (1 - β[2]) * abs2(dx)
287 ρ = ρ∞ - 2*t * βt[2] / (1 - βt[2]) |> real
288 if ρ > 4

293 end
294
295 return (mt, vt, βt .* β, t + 1), dx′
296 end

501 """
502 AdamW(η = 0.001, β = (0.9, 0.999), λ = 0, ϵ = 1e-8; couple = true)
503 AdamW(; [eta, beta, lambda, epsilon, couple])
504
505 [AdamW](https://arxiv.org/abs/1711.05101) is a variant of Adam fixing (as in repairing) its
506 weight decay regularization.
507 Implemented as an [`OptimiserChain`](@ref) of [`Adam`](@ref) and [`WeightDecay`](@ref)`.
508
509 # Parameters
510 - Learning rate (`η == eta`): Amount by which gradients are discounted before updating
511 the weights.
512 - Decay of momentums (`β::Tuple == beta`): Exponential decay for the first (β1) and the
513 second (β2) momentum estimate.
514 - Weight decay (`λ == lambda`): Controls the strength of ``L_2`` regularisation.
515 - Machine epsilon (`ϵ == epsilon`): Constant to prevent division by zero
516 (no need to change default)
517 - Keyword `couple`: If `true`, the weight decay is coupled with the learning rate, as in pytorch's AdamW.
518 This corresponds to an update of the form `x = x - η * (dx + λ * x)`, where `dx` is the
519 update from Adam with learning rate 1.
520 If `false`, the weight decay is decoupled from the learning rate, in the spirit of the original paper.
521 This corresponds to an update of the form `x = x - η * dx - λ * x`.
522 Default is `true`.
523
524 !!! warning "Breaking change in v0.4"
525 With version 0.4 the default update rule for AdamW has changed to match the pytorch implementation.
526 The previous rule, which is closer to the original paper, can be obtained by setting `AdamW(..., couple=false)`.
527 See [this issue](https://github.com/FluxML/Flux.jl/issues/2433) for more details.
528 """
529 struct AdamW{Teta,Tbeta<:Tuple,Tlambda,Teps} <: AbstractRule
530 eta::Teta
531 beta::Tbeta
532 lambda::Tlambda
533 epsilon::Teps
534 couple::Bool
535 end
536
537 function AdamW(η, β = (0.9, 0.999), λ = 0.0, ϵ = 1e-8; couple::Bool = true)
538 η < 0 && throw(DomainError(η, "the learning rate cannot be negative"))
539 return AdamW(float(η), β, float(λ), float(ϵ), couple)
540 end

546
547 function apply!(o::AdamW, state, x::AbstractArray{T}, dx) where T
548 η, β, ϵ, λ = T(o.eta), T.(o.beta), T(o.epsilon), T(o.lambda)
549 mt, vt, βt = state
550
551 # standard Adam update with learning rate eta=1
552 @.. mt = β[1] * mt + (1 - β[1]) * dx
553 @.. vt = β[2] * vt + (1 - β[2]) * abs2(dx)
554 dx′ = @lazy mt / (1 - βt[1]) / (sqrt(vt / (1 - βt[2])) + ϵ)
555
556 # apply learning rate and weight decay
557 if o.couple
558 dx′′ = @lazy η * (dx′ + λ * x)
559 else
560 dx′′ = @lazy η * dx′ + λ * x
561 end
562
563 return (mt, vt, βt .* β), dx′′
564 end

595 η, β, ϵ = T(o.eta), T.(o.beta), _eps(T, o.epsilon)
596 mt, st, βt = state
597
598 @.. mt = β[1] * mt + (1 - β[1]) * dx
599 @.. st = β[2] * st + (1 - β[2]) * abs2(dx - mt) + ϵ
600 dx′ = @lazy η * mt / (1 - βt[1]) / (sqrt(st / (1 - βt[2])) + ϵ)
601
602 return (mt, st, βt .* β), dx′
603 end

697 """
698 ClipNorm(ω = 10, p = 2; throw = true)
699
700 Scales any gradient array for which `norm(dx, p) > ω`
701 to stay at this threshold (unless `p==0`).
702
703 Throws an error if the norm is infinite or `NaN`,
704 which you can turn off with `throw = false`.
705
706 Typically composed with other rules using [`OptimiserChain`](@ref).
707
708 See also [`ClipGrad`](@ref).
709 """
710 struct ClipNorm{To,Tp} <: AbstractRule
711 omega::To
712 p::Tp
713 throw::Bool
714 end
715 ClipNorm(ω = 10, p = 2; throw::Bool = true) = ClipNorm(float(ω), float(p), throw)
716
717 init(o::ClipNorm, x::AbstractArray) = nothing
718
719 function apply!(o::ClipNorm, state, x::AbstractArray{T}, dx) where T
720 nrm = _norm(dx, o.p)
721 if o.throw && !isfinite(nrm)
722 throw(DomainError("gradient has $(o.p)-norm $nrm, for array $(summary(x))"))
723 end
724 λ = T(min(o.omega / nrm, 1))
725
726 return state, @lazy dx * λ
727 end
728
729 _norm(dx::AbstractArray, p::Real) = norm(dx, p) # LinearAlgebra, CUDA
730 function _norm(dx::Broadcast.Broadcasted, p::Real)
731 if p == 2
732 # This lacks the undeflow/overflow tests of LinearAlgebra's version
733 sqrt(sum(abs2, dx))
734 elseif p == 1
735 float(sum(abs, dx))
736 elseif p == Inf
737 float(maximum(abs, dx))
738 elseif p == 0
739 cnt = count(!iszero, dx)
740 T = Base.@default_eltype dx
741 T <: Number ? convert(float(T), cnt) : cnt
742 elseif p == -Inf
743 float(minimum(abs, dx))
744 else
745 # This isn't optimally fast but does ensure p::Float64 doesn't promote
746 tmp = abs.(dx)
747 q = convert(float(eltype(tmp)), p)
748 sum(tmp .^ q) ^ (1/q)
749 end
750 end

752 """
753 OptimiserChain(opts...)
754
755 Compose a sequence of optimisers so that each `opt` in `opts`
756 updates the gradient, in the order specified.
757
758 With an empty sequence, `OptimiserChain()` is the identity,
759 so `update!` will subtract the full gradient from the parameters.
760 This is equivalent to `Descent(1)`.
761
762 # Example
763
764 ```jldoctest
765 julia> o = OptimiserChain(ClipGrad(1.0), Descent(0.1));
766
767 julia> m = (zeros(3),);
768
769 julia> s = Optimisers.setup(o, m)
770 (Leaf(OptimiserChain(ClipGrad(1.0), Descent(0.1)), (nothing, nothing)),)
771
772 julia> Optimisers.update(s, m, ([0.3, 1, 7],))[2] # clips before discounting
773 ([-0.03, -0.1, -0.1],)
774 ```
775 """
776 struct OptimiserChain{O<:Tuple} <: AbstractRule
777 opts::O
778 end
779 OptimiserChain(opts...) = OptimiserChain(opts)
780
781 @functor OptimiserChain
782
783 init(o::OptimiserChain, x::AbstractArray) = map(opt -> init(opt, x), o.opts)
784
785 function apply!(o::OptimiserChain, states, x, dx, dxs...)
786 foldl(tuple.(o.opts, states); init = ((), dx)) do (states′, dx′), (opt, state)
787 if dx′ isa Zero
788 return (states′..., state), dx′
789 else
790 state′, dx′ = apply!(opt, state, x, dx′, dxs...)
791 return (states′..., state′), dx′
792 end
793 end
794 end
795
796 function Base.show(io::IO, c::OptimiserChain)
797 print(io, "OptimiserChain(")
798 join(io, c.opts, ", ")
799 print(io, ")")
800 end
801
802 adjust(ℓ::OptimiserChain, eta::Real) = OptimiserChain(map(opt -> adjust(opt, eta), ℓ.opts)...)
803 adjust(ℓ::OptimiserChain; kw...) = OptimiserChain(map(opt -> adjust(opt; kw...), ℓ.opts)...)

806 """
807 AccumGrad(n::Int)
808
809 A rule constructed `OptimiserChain(AccumGrad(n), Rule())` will accumulate for `n` steps,
810 before applying `Rule` to the mean of these `n` gradients.
811
812 This is useful for training with effective batch sizes too large for the available memory.
813 Instead of computing the gradient for batch size `b` at once, compute it for size `b/n` and
814 accumulate `n` such gradients.
815
816 # Example
817 ```jldoctest
818 julia> m = (x=[1f0], y=[2f0]);
819
820 julia> r = OptimiserChain(AccumGrad(2), WeightDecay(0.01), Descent(0.1));
821
822 julia> s = Optimisers.setup(r, m);
823
824 julia> Optimisers.update!(s, m, (x=[33], y=[0]));
825
826 julia> m # model not yet changed
827 (x = Float32[1.0], y = Float32[2.0])
828
829 julia> Optimisers.update!(s, m, (x=[0], y=[444]));
830
831 julia> m # n=2 gradients applied at once
832 (x = Float32[-0.651], y = Float32[-20.202002])
833 ```
834 """
835 struct AccumGrad{N} <: AbstractRule
836 n::N
837 end
838
839 function AccumGrad(n::Integer)
840 n > 0 || throw(ArgumentError("AccumGrad must accumulate at least one gradient"))
841 return AccumGrad{typeof(n)}(n)
842 end
843
844 init(::AccumGrad{N}, x) where N = (zero(x), N(1))
845
846 function apply!(o::AccumGrad, state, x, dx)
847 accum_dx, counter = state
848 if counter == 1
849 @.. accum_dx = dx / o.n
850 else
851 @.. accum_dx = accum_dx + dx / o.n
852 end
853 if counter == o.n
854 return (accum_dx, 1), accum_dx
855 else
856 return (accum_dx, counter + 1), nothing
857 end
858 end
