100 """
101 lookup(pointer::Ptr{Cvoid}) -> Vector{StackFrame}
102
103 Given a pointer to an execution context (usually generated by a call to `backtrace`), looks
104 up stack frame context information. Returns an array of frame information for all functions
105 inlined at that point, innermost function first.
106 """
107 Base.@constprop :none function lookup(pointer::Ptr{Cvoid})
108 infos = ccall(:jl_lookup_code_address, Any, (Ptr{Cvoid}, Cint), pointer, false)::Core.SimpleVector
109 pointer = convert(UInt64, pointer)
110 isempty(infos) && return [StackFrame(empty_sym, empty_sym, -1, nothing, true, false, pointer)] # this is equal to UNKNOWN
111 res = Vector{StackFrame}(undef, length(infos))
112 for i in 1:length(infos)
113 info = infos[i]::Core.SimpleVector
114 @assert(length(info) == 6)
115 res[i] = StackFrame(info[1]::Symbol, info[2]::Symbol, info[3]::Int, info[4], info[5]::Bool, info[6]::Bool, pointer)
116 end
117 return res
118 end

120 const top_level_scope_sym = Symbol("top-level scope")
121
122 function lookup(ip::Union{Base.InterpreterIP,Core.Compiler.InterpreterIP})
123 code = ip.code
124 if code === nothing
125 # interpreted top-level expression with no CodeInfo
126 return [StackFrame(top_level_scope_sym, empty_sym, 0, nothing, false, false, 0)]
127 end
128 codeinfo = (code isa MethodInstance ? code.uninferred : code)::CodeInfo
129 # prepare approximate code info
130 if code isa MethodInstance && (meth = code.def; meth isa Method)
131 func = meth.name
132 file = meth.file
133 line = meth.line
134 else
135 func = top_level_scope_sym
136 file = empty_sym
137 line = Int32(0)
138 end
139 i = max(ip.stmt+1, 1) # ip.stmt is 0-indexed
140 if i > length(codeinfo.codelocs) || codeinfo.codelocs[i] == 0
141 return [StackFrame(func, file, line, code, false, false, 0)]
142 end
143 lineinfo = codeinfo.linetable[codeinfo.codelocs[i]]::Core.LineInfoNode
144 scopes = StackFrame[]
145 while true
146 inlined = lineinfo.inlined_at != 0
147 push!(scopes, StackFrame(Base.IRShow.method_name(lineinfo)::Symbol, lineinfo.file, lineinfo.line, inlined ? nothing : code, false, inlined, 0))
148 inlined || break
149 lineinfo = codeinfo.linetable[lineinfo.inlined_at]::Core.LineInfoNode
150 end
151 return scopes
152 end

154 """
155 stacktrace([trace::Vector{Ptr{Cvoid}},] [c_funcs::Bool=false]) -> StackTrace
156
157 Return a stack trace in the form of a vector of `StackFrame`s. (By default stacktrace
158 doesn't return C functions, but this can be enabled.) When called without specifying a
159 trace, `stacktrace` first calls `backtrace`.
160 """
161 Base.@constprop :none function stacktrace(trace::Vector{<:Union{Base.InterpreterIP,Core.Compiler.InterpreterIP,Ptr{Cvoid}}}, c_funcs::Bool=false)
162 stack = StackTrace()
163 for ip in trace
164 for frame in lookup(ip)
165 # Skip frames that come from C calls.
166 if c_funcs || !frame.from_c
167 push!(stack, frame)
168 end
169 end
170 end
171 return stack
172 end

174 Base.@constprop :none function stacktrace(c_funcs::Bool=false)
175 stack = stacktrace(backtrace(), c_funcs)
176 # Remove frame for this function (and any functions called by this function).
177 remove_frames!(stack, :stacktrace)
178 # also remove all of the non-Julia functions that led up to this point (if that list is non-empty)
179 c_funcs && deleteat!(stack, 1:(something(findfirst(frame -> !frame.from_c, stack), 1) - 1))
180 return stack
181 end

183 """
184 remove_frames!(stack::StackTrace, name::Symbol)
185
186 Takes a `StackTrace` (a vector of `StackFrames`) and a function name (a `Symbol`) and
187 removes the `StackFrame` specified by the function name from the `StackTrace` (also removing
188 all frames above the specified function). Primarily used to remove `StackTraces` functions
189 from the `StackTrace` prior to returning it.
190 """
191 function remove_frames!(stack::StackTrace, name::Symbol)
192 deleteat!(stack, 1:something(findlast(frame -> frame.func == name, stack), 0))
193 return stack
194 end

196 function remove_frames!(stack::StackTrace, names::Vector{Symbol})
197 deleteat!(stack, 1:something(findlast(frame -> frame.func in names, stack), 0))
198 return stack

213 function show_spec_linfo(io::IO, frame::StackFrame)
214 linfo = frame.linfo
215 if linfo === nothing
216 if frame.func === empty_sym
217 print(io, "ip:0x", string(frame.pointer, base=16))
218 elseif frame.func === top_level_scope_sym
219 print(io, "top-level scope")
220 else
221 Base.print_within_stacktrace(io, Base.demangle_function_name(string(frame.func)), bold=true)
222 end
223 elseif linfo isa CodeInfo
224 print(io, "top-level scope")
225 elseif linfo isa Module
226 Base.print_within_stacktrace(io, Base.demangle_function_name(string(frame.func)), bold=true)
227 elseif linfo isa MethodInstance
228 def = linfo.def
229 if def isa Module
230 Base.show_mi(io, linfo, #=from_stackframe=#true)
231 else
232 show_spec_sig(io, def, linfo.specTypes)
233 end
234 else
235 m = linfo::Method
236 show_spec_sig(io, m, m.sig)
237 end
238 end

240 function show_spec_sig(io::IO, m::Method, @nospecialize(sig::Type))
241 if get(io, :limit, :false)::Bool
242 if !haskey(io, :displaysize)
243 io = IOContext(io, :displaysize => displaysize(io))
244 end
245 end
246 argnames = Base.method_argnames(m)
247 argnames = replace(argnames, :var"#unused#" => :var"")
248 if m.nkw > 0
249 # rearrange call kw_impl(kw_args..., func, pos_args...) to func(pos_args...; kw_args)
250 kwarg_types = Any[ fieldtype(sig, i) for i = 2:(1+m.nkw) ]
251 uw = Base.unwrap_unionall(sig)::DataType
252 pos_sig = Base.rewrap_unionall(Tuple{uw.parameters[(m.nkw+2):end]...}, sig)
253 kwnames = argnames[2:(m.nkw+1)]
254 for i = 1:length(kwnames)
255 str = string(kwnames[i])::String
256 if endswith(str, "...")
257 kwnames[i] = Symbol(str[1:end-3])
258 end
259 end
260 Base.show_tuple_as_call(io, m.name, pos_sig;
261 demangle=true,
262 kwargs=zip(kwnames, kwarg_types),
263 argnames=argnames[m.nkw+2:end])
264 else
265 Base.show_tuple_as_call(io, m.name, sig; demangle=true, argnames)
266 end

269 function show(io::IO, frame::StackFrame)
270 show_spec_linfo(io, frame)
271 if frame.file !== empty_sym
272 file_info = basename(string(frame.file))
273 print(io, " at ")
274 print(io, file_info, ":")

278 print(io, "?")

286 function Base.parentmodule(frame::StackFrame)
287 linfo = frame.linfo
288 if linfo isa MethodInstance
