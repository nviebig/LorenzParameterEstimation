80 MersenneTwister(seed)

82
83 Create a `MersenneTwister` RNG object. Different RNG objects can have
84 their own seeds, which may be useful for generating different streams
85 of random numbers.
86 The `seed` may be an integer, a string, or a vector of `UInt32` integers.
87 If no seed is provided, a randomly generated one is created (using entropy from the system).
88 See the [`seed!`](@ref) function for reseeding an already existing `MersenneTwister` object.

101
102 julia> x2 = rand(MersenneTwister(123), 2)
103 2-element Vector{Float64}:

105 0.8735343642013971

283 # random_seed tries to produce a random seed of type UInt128 from system entropy
284 function random_seed()
285 try
286 # as MersenneTwister prints its seed when `show`ed, 128 bits is a good compromise for
287 # almost surely always getting distinct seeds, while having them printed reasonably tersely
288 return rand(RandomDevice(), UInt128)
289 catch ex

292 return Libc.rand()
293 end

296 function hash_seed(seed::Integer)
297 ctx = SHA.SHA2_256_CTX()
298 neg = signbit(seed)
299 if neg
300 seed = ~seed
301 end
302 @assert seed >= 0
303 while true
304 word = (seed % UInt32) & 0xffffffff
305 seed >>>= 32
306 SHA.update!(ctx, reinterpret(NTuple{4, UInt8}, word))
307 iszero(seed) && break
308 end
309 # make sure the hash of negative numbers is different from the hash of positive numbers
310 neg && SHA.update!(ctx, (0x01,))
311 SHA.digest!(ctx)
312 end

314 function hash_seed(seed::Union{AbstractArray{UInt32}, AbstractArray{UInt64}})
315 ctx = SHA.SHA2_256_CTX()
316 for xx in seed
317 SHA.update!(ctx, reinterpret(NTuple{8, UInt8}, UInt64(xx)))
318 end
319 # discriminate from hash_seed(::Integer)
320 SHA.update!(ctx, (0x10,))
321 SHA.digest!(ctx)
322 end

324 function hash_seed(str::AbstractString)
325 ctx = SHA.SHA2_256_CTX()
326 # convert to String such that `codeunits(str)` below is consistent between equal
327 # strings of different types
328 str = String(str)
329 SHA.update!(ctx, codeunits(str))
330 # signature for strings: so far, all hash_seed functions end-up hashing a multiple
331 # of 4 bytes of data, and add the signature (1 byte) at the end; so hash as many
332 # bytes as necessary to have a total number of hashed bytes equal to 0 mod 4 (padding),
333 # and then hash the signature 0x05; in order for strings of different lengths to have
334 # different hashes, padding bytes are set equal to the number of padding bytes
335 pad = 4 - mod(ncodeunits(str), 4)
336 for _=1:pad
337 SHA.update!(ctx, (pad % UInt8,))
338 end
339 SHA.update!(ctx, (0x05,))
340 SHA.digest!(ctx)
341 end

361 function initstate!(r::MersenneTwister, data::StridedVector, seed)
362 # we deepcopy `seed` because the caller might mutate it, and it's useful
363 # to keep it constant inside `MersenneTwister`; but multiple instances
364 # can share the same seed without any problem (e.g. in `copy`)
365 r.seed = deepcopy(seed)
366 dsfmt_init_by_array(r.state, reinterpret(UInt32, data))
367 reset_caches!(r)
368 r.adv = 0
369 r.adv_jump = 0
370 return r
371 end
372
373 # when a seed is not provided, we generate one via `RandomDevice()` in `random_seed()` rather
374 # than calling directly `initstate!` with `rand(RandomDevice(), UInt32, whatever)` because the
375 # seed is printed in `show(::MersenneTwister)`, so we need one; the cost of `hash_seed` is a
376 # small overhead compared to `initstate!`, so this simple solution is fine
377 seed!(r::MersenneTwister, ::Nothing) = seed!(r, random_seed())
378 seed!(r::MersenneTwister, seed) = initstate!(r, hash_seed(seed), seed)

383 """
384 Random.default_rng() -> rng
385
386 Return the default global random number generator (RNG), which is used by `rand`-related functions when
387 no explicit RNG is provided.
388
389 When the `Random` module is loaded, the default RNG is _randomly_ seeded, via [`Random.seed!()`](@ref):
390 this means that each time a new julia session is started, the first call to `rand()` produces a different
391 result, unless `seed!(seed)` is called first.
392
393 It is thread-safe: distinct threads can safely call `rand`-related functions on `default_rng()` concurrently,
394 e.g. `rand(default_rng())`.
395
396 !!! note
397 The type of the default RNG is an implementation detail. Across different versions of
398 Julia, you should not expect the default RNG to always have the same type, nor that it will
399 produce the same stream of random numbers for a given seed.
400
401 !!! compat "Julia 1.3"
402 This function was introduced in Julia 1.3.
403 """
404 @inline default_rng() = TaskLocalRNG()
405 @inline default_rng(tid::Int) = TaskLocalRNG()

566 function rand!(r::MersenneTwister, A::UnsafeView{Float64},
567 I::SamplerTrivial{<:FloatInterval_64})
568 # depending on the alignment of A, the data written by fill_array! may have
569 # to be left-shifted by up to 15 bytes (cf. unsafe_copyto! below) for
570 # reproducibility purposes;
571 # so, even for well aligned arrays, fill_array! is used to generate only
572 # the n-2 first values (or n-3 if n is odd), and the remaining values are
573 # generated by the scalar version of rand
574 n = length(A)
575 n2 = (n-2) ÷ 2 * 2
576 n2 < dsfmt_get_min_array_size() && return _rand_max383!(r, A, I[])
577
578 pA = A.ptr
579 align = Csize_t(pA) % 16
580 if align > 0
581 pA2 = pA + 16 - align
582 fill_array!(r, pA2, n2, I[]) # generate the data in-place, but shifted
583 unsafe_copyto!(pA, pA2, n2) # move the data to the beginning of the array
584 else
585 fill_array!(r, pA, n2, I[])
586 end
587 for i=n2+1:n
588 A[i] = rand(r, I[])
589 end
590 A
591 end

595 # n64 is the length in terms of `Float64` of the target
596 @assert sizeof(Float64)*n64 <= sizeof(T)*length(A) && isbitstype(T)
597 GC.@preserve A rand!(r, UnsafeView{Float64}(pointer(A), n64), SamplerTrivial(I))
598 A
599 end

601 ##### Array: Float64, Float16, Float32
602
603 rand!(r::MersenneTwister, A::Array{Float64}, I::SamplerTrivial{<:FloatInterval_64}) =
604 _rand!(r, A, length(A), I[])
605
606 mask128(u::UInt128, ::Type{Float16}) =
607 (u & 0x03ff03ff03ff03ff03ff03ff03ff03ff) | 0x3c003c003c003c003c003c003c003c00
608
609 mask128(u::UInt128, ::Type{Float32}) =
610 (u & 0x007fffff007fffff007fffff007fffff) | 0x3f8000003f8000003f8000003f800000
611
612 for T in (Float16, Float32)
613 @eval function rand!(r::MersenneTwister, A::Array{$T}, ::SamplerTrivial{CloseOpen12{$T}})
614 n = length(A)
615 n128 = n * sizeof($T) ÷ 16
616 _rand!(r, A, 2*n128, CloseOpen12())
617 GC.@preserve A begin
618 A128 = UnsafeView{UInt128}(pointer(A), n128)
619 for i in 1:n128
620 u = A128[i]
621 u ⊻= u << 26
622 # at this point, the 64 low bits of u, "k" being the k-th bit of A128[i] and "+"
623 # the bit xor, are:
624 # [..., 58+32,..., 53+27, 52+26, ..., 33+7, 32+6, ..., 27+1, 26, ..., 1]
625 # the bits needing to be random are
626 # [1:10, 17:26, 33:42, 49:58] (for Float16)
627 # [1:23, 33:55] (for Float32)
628 # this is obviously satisfied on the 32 low bits side, and on the high side,
629 # the entropy comes from bits 33:52 of A128[i] and then from bits 27:32
630 # (which are discarded on the low side)
631 # this is similar for the 64 high bits of u
632 A128[i] = mask128(u, $T)
633 end
634 end
635 for i in 16*n128÷sizeof($T)+1:n
636 @inbounds A[i] = rand(r, $T) + one($T)
637 end
638 A
639 end

641 @eval function rand!(r::MersenneTwister, A::Array{$T}, ::SamplerTrivial{CloseOpen01{$T}})
642 rand!(r, A, CloseOpen12($T))
643 I32 = one(Float32)
644 for i in eachindex(A)
645 @inbounds A[i] = Float32(A[i])-I32 # faster than "A[i] -= one(T)" for T==Float16
646 end
647 A
648 end
649 end

653 function rand!(r::MersenneTwister, A::UnsafeView{UInt128}, ::SamplerType{UInt128})
654 n::Int=length(A)
655 i = n
656 while true
657 rand!(r, UnsafeView{Float64}(A.ptr, 2i), CloseOpen12())
658 n < 5 && break
659 i = 0
660 while n-i >= 5
661 u = A[i+=1]
662 A[n] ⊻= u << 48
663 A[n-=1] ⊻= u << 36
664 A[n-=1] ⊻= u << 24
665 A[n-=1] ⊻= u << 12
666 n-=1
667 end
668 end
669 if n > 0
670 u = rand(r, UInt2x52Raw())
671 for i = 1:n
672 A[i] ⊻= u << (12*i)
673 end
674 end
675 A
676 end

700 # similar to Array{UInt8}, but we need to mask the result so that only the LSB
701 # in each byte can be non-zero
702
703 function rand!(r::MersenneTwister, A1::Array{Bool}, sp::SamplerType{Bool})
704 n1 = length(A1)
705 n128 = n1 ÷ 16
706
707 if n128 == 0
708 bits = rand(r, UInt52Raw())
709 else
710 GC.@preserve A1 begin
711 A = UnsafeView{UInt128}(pointer(A1), n128)
712 rand!(r, UnsafeView{Float64}(A.ptr, 2*n128), CloseOpen12())
713 # without masking, non-zero bits could be observed in other
714 # positions than the LSB of each byte
715 mask = 0x01010101010101010101010101010101
716 # we need up to 15 bits of entropy in `bits` for the final loop,
717 # which we will extract from x = A[1] % UInt64;
718 # let y = x % UInt32; y contains 32 bits of entropy, but 4
719 # of them will be used for A[1] itself (the first of
720 # each byte). To compensate, we xor with (y >> 17),
721 # which gets the entropy from the second bit of each byte
722 # of the upper-half of y, and sets it in the first bit
723 # of each byte of the lower half; the first two bytes
724 # now contain 16 usable random bits
725 x = A[1] % UInt64
726 bits = x ⊻ x >> 17
727 for i = 1:n128
728 # << 5 to randomize the first bit of the 8th & 16th byte
729 # (i.e. we move bit 52 (resp. 52 + 64), which is unused,
730 # to position 57 (resp. 57 + 64))
731 A[i] = (A[i] ⊻ A[i] << 5) & mask
732 end
733 end
734 end
735 for i = 16*n128+1:n1
736 @inbounds A1[i] = bits % Bool
737 bits >>= 1
738 end
739 A1
740 end

757 # NON-PUBLIC
758 function jump(r::MersenneTwister, steps::Integer)
759 iseven(steps) || throw(DomainError(steps, "steps must be even"))
760 # steps >= 0 checked in calc_jump (`steps >> 1 < 0` if `steps < 0`)
761 j = _randjump(r, Random.DSFMT.calc_jump(steps >> 1))
762 j.adv_jump += steps
763 j
764 end

787 # advances raw state (per fill_array!) of r by n steps (Float64 values)
788 function _advance_n!(r::MersenneTwister, n::Int64, work::Vector{Float64})
789 n == 0 && return
790 n < 0 && throw(DomainError(n, "can't advance $r to the specified state"))
791 ms = dsfmt_get_min_array_size() % Int64
792 @assert n >= ms
793 lw = ms + n % ms
794 resize!(work, lw)
795 GC.@preserve work fill_array!(r, pointer(work), lw, CloseOpen12())
796 c::Int64 = lw
797 GC.@preserve work while n > c
798 fill_array!(r, pointer(work), ms, CloseOpen12())
799 c += ms
800 end
801 @assert n == c
802 end
803
804 function _advance_to!(r::MersenneTwister, adv::Int64, work)
805 _advance_n!(r, adv - r.adv, work)
806 @assert r.adv == adv
807 end

809 function _advance_F!(r::MersenneTwister, adv_vals, idxF, work)
810 _advance_to!(r, adv_vals, work)
811 gen_rand(r)
812 @assert r.adv_vals == adv_vals
813 r.idxF = idxF
814 end

816 function _advance_I!(r::MersenneTwister, adv_ints, idxI, work)
817 _advance_to!(r, adv_ints, work)
818 mt_setfull!(r, Int) # sets r.adv_ints
819 @assert r.adv_ints == adv_ints
820 r.idxI = 16*length(r.ints) - 8*idxI
821 end

823 function advance!(r::MersenneTwister, adv_jump, adv, adv_vals, idxF, adv_ints, idxI)
824 adv_jump = BigInt(adv_jump)
825 adv, adv_vals, adv_ints = Int64.((adv, adv_vals, adv_ints))
826 idxF, idxI = Int.((idxF, idxI))
827
828 ms = dsfmt_get_min_array_size() % Int
829 work = sizehint!(Vector{Float64}(), 2ms)
830
831 adv_jump != 0 && jump!(r, adv_jump)
832 advF = (adv_vals, idxF) != (0, 0)
833 advI = (adv_ints, idxI) != (0, 0)
834
835 if advI && advF
836 @assert adv_vals != adv_ints
837 if adv_vals < adv_ints
838 _advance_F!(r, adv_vals, idxF, work)
839 _advance_I!(r, adv_ints, idxI, work)
840 else
841 _advance_I!(r, adv_ints, idxI, work)
842 _advance_F!(r, adv_vals, idxF, work)
843 end
844 elseif advF
845 _advance_F!(r, adv_vals, idxF, work)
846 elseif advI
847 _advance_I!(r, adv_ints, idxI, work)
848 else
849 @assert adv == 0
850 end
851 _advance_to!(r, adv, work)
852 r
853 end
