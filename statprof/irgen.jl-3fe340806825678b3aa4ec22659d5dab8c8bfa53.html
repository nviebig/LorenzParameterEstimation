129 # this pass lowers `jl_throw` and friends to GPU-compatible exceptions.
130 # this isn't strictly necessary, but has a couple of advantages:
131 # - we can kill off unused exception arguments that otherwise would allocate or invoke
132 # - we can fake debug information (lacking a stack unwinder)
133 #
134 # once we have thorough inference (ie. discarding `@nospecialize` and thus supporting
135 # exception arguments) and proper debug info to unwind the stack, this pass can go.
136 function lower_throw!(mod::LLVM.Module)

266 # Determine the calling convention of a the arguments of a Julia function, given the
267 # LLVM function type as generated by the Julia code generator. Returns an vector with one
268 # element for each Julia-level argument, containing a tuple with the following fields:
269 # - `cc`: the calling convention of the argument
270 # - `typ`: the Julia type of the argument
271 # - `name`: the name of the argument
272 # - `idx`: the index of the argument in the LLVM function type, or `nothing` if the argument
273 # is not passed at the LLVM level.
274 function classify_arguments(@nospecialize(job::CompilerJob), codegen_ft::LLVM.FunctionType;
275 post_optimization::Bool=false)
276 source_sig = job.source.specTypes
277 source_types = [source_sig.parameters...]
278
279 source_argnames = Base.method_argnames(job.source.def)
280 while length(source_argnames) < length(source_types)
281 # this is probably due to a trailing vararg; repeat its name
282 push!(source_argnames, source_argnames[end])
283 end
284
285 codegen_types = parameters(codegen_ft)
286
287 if post_optimization && kernel_state_type(job) !== Nothing
288 args = []
289 push!(args, (cc=KERNEL_STATE, typ=kernel_state_type(job), name=:kernel_state, idx=1))
290 codegen_i = 2
291 else
292 args = []
293 codegen_i = 1
294 end
295 for (source_typ, source_name) in zip(source_types, source_argnames)
296 if isghosttype(source_typ) || Core.Compiler.isconstType(source_typ)
297 push!(args, (cc=GHOST, typ=source_typ, name=source_name, idx=nothing))
298 continue
299 end
300
301 codegen_typ = codegen_types[codegen_i]
302
303 if codegen_typ isa LLVM.PointerType
304 llvm_source_typ = convert(LLVMType, source_typ; allow_boxed=true)
305 # pointers are used for multiple kinds of arguments
306 # - literal pointer values
307 if source_typ <: Ptr || source_typ <: Core.LLVMPtr
308 @assert llvm_source_typ == codegen_typ
309 push!(args, (cc=BITS_VALUE, typ=source_typ, name=source_name, idx=codegen_i))
310 # - boxed values
311 # XXX: use `deserves_retbox` instead?
312 elseif llvm_source_typ isa LLVM.PointerType
313 @assert llvm_source_typ == codegen_typ
314 push!(args, (cc=MUT_REF, typ=source_typ, name=source_name, idx=codegen_i))
315 # - references to aggregates
316 else
317 @assert llvm_source_typ != codegen_typ
318 push!(args, (cc=BITS_REF, typ=source_typ, name=source_name, idx=codegen_i))
319 end
320 else
321 push!(args, (cc=BITS_VALUE, typ=source_typ, name=source_name, idx=codegen_i))
322 end
323
324 codegen_i += 1
325 end
326
327 return args
328 end

495 # kernel state arguments
496 #
497 # to facilitate passing stateful information to kernels without having to recompile, e.g.,
498 # the storage location for exception flags, or the location of a I/O buffer, we enable the
499 # back-end to specify a Julia object that will be passed to the kernel by-value, and to
500 # every called function by-reference. Access to this object is done using the
501 # `julia.gpu.state_getter` intrinsic. after optimization, these intrinsics will be lowered
502 # to refer to the state argument.
503 #
504 # note that we deviate from the typical Julia calling convention, by always passing the
505 # state objects by value instead of by reference, this to ensure that the state object
506 # is not copied to the stack (because LLVM doesn't see that all uses are read-only).
507 # in principle, `readonly byval` should be equivalent, but LLVM doesn't realize that.
508 # also see https://github.com/JuliaGPU/CUDA.jl/pull/1167 and the comments in that PR.
509 # once LLVM supports this pattern, consider going back to passing the state by reference,
510 # so that the julia.gpu.state_getter` can be simplified to return an opaque pointer.
511
512 # add a state argument to every function in the module, starting from the kernel entry point
513 function add_kernel_state!(mod::LLVM.Module)
514 job = current_job::CompilerJob
515
516 # check if we even need a kernel state argument
517 state = kernel_state_type(job)
518 @assert job.config.kernel
519 if state === Nothing
520 return false
521 end
522 T_state = convert(LLVMType, state)
523
524 # intrinsic returning an opaque pointer to the kernel state.
525 # this is both for extern uses, and to make this transformation a two-step process.
526 state_intr = kernel_state_intr(mod, T_state)
527 state_intr_ft = LLVM.FunctionType(T_state)
528
529 # determine which functions need a kernel state argument
530 #
531 # previously, we add the argument to every function and relied on unused arg elim to
532 # clean-up the IR. however, some libraries do Funny Stuff, e.g., libdevice bitcasting
533 # function pointers. such IR is hard to rewrite, so instead be more conservative.
534 worklist = Set{LLVM.Function}([state_intr, kernels(mod)...])
535 worklist_length = 0
536 while worklist_length != length(worklist)
537 # iteratively discover functions that use the intrinsic or any function calling it
538 worklist_length = length(worklist)
539 additions = LLVM.Function[]
540 function check_user(val)
541 if val isa Instruction
542 bb = LLVM.parent(val)
543 new_f = LLVM.parent(bb)
544 in(new_f, worklist) || push!(additions, new_f)
545 elseif val isa ConstantExpr
546 # constant expressions don't have a parent; we need to look up their uses
547 for use in uses(val)
548 check_user(user(use))
549 end
550 else
551 error("Don't know how to check uses of $val. Please file an issue.")
552 end
553 end
554 for f in worklist, use in uses(f)
555 check_user(user(use))
556 end
557 for f in additions
558 push!(worklist, f)
559 end
560 end
561 delete!(worklist, state_intr)
562
563 # add a state argument
564 workmap = Dict{LLVM.Function, LLVM.Function}()
565 for f in worklist
566 fn = LLVM.name(f)
567 ft = function_type(f)
568 LLVM.name!(f, fn * ".stateless")
569
570 # create a new function
571 new_param_types = [T_state, parameters(ft)...]
572 new_ft = LLVM.FunctionType(return_type(ft), new_param_types)
573 new_f = LLVM.Function(mod, fn, new_ft)
574 LLVM.name!(parameters(new_f)[1], "state")
575 linkage!(new_f, linkage(f))
576 for (arg, new_arg) in zip(parameters(f), parameters(new_f)[2:end])
577 LLVM.name!(new_arg, LLVM.name(arg))
578 end
579
580 workmap[f] = new_f
581 end
582
583 # clone and rewrite the function bodies, replacing uses of the old stateless function
584 # with the newly created definition that includes the state argument.
585 #
586 # most uses are rewritten by LLVM by putting the functions in the value map.
587 # a separate value materializer is used to recreate constant expressions.
588 #
589 # note that this only _replaces_ the uses of these functions, we'll still need to
590 # _correct_ the uses (i.e. actually add the state argument) afterwards.
591 function materializer(val)
592 if val isa ConstantExpr
593 if opcode(val) == LLVM.API.LLVMBitCast
594 target = operands(val)[1]
595 if target isa LLVM.Function && haskey(workmap, target)
596 # the function is being bitcasted to a different function type.
597 # we need to mutate that function type to include the state argument,
598 # or we'd be invoking the original function in an invalid way.
599 #
600 # XXX: ptrtoint/inttoptr pairs can also lose the state argument...
601 # is all this even sound?
602 typ = value_type(val)::LLVM.PointerType
603 ft = eltype(typ)::LLVM.FunctionType
604 new_ft = LLVM.FunctionType(return_type(ft), [T_state, parameters(ft)...])
605 return const_bitcast(workmap[target], LLVM.PointerType(new_ft, addrspace(typ)))
606 end
607 elseif opcode(val) == LLVM.API.LLVMPtrToInt
608 target = operands(val)[1]
609 if target isa LLVM.Function && haskey(workmap, target)
610 return const_ptrtoint(workmap[target], value_type(val))
611 end
612 end
613 end
614 return nothing # do not claim responsibility
615 end
616 for (f, new_f) in workmap
617 # use a value mapper for rewriting function arguments
618 value_map = Dict{LLVM.Value, LLVM.Value}()
619 for (param, new_param) in zip(parameters(f), parameters(new_f)[2:end])
620 LLVM.name!(new_param, LLVM.name(param))
621 value_map[param] = new_param
622 end

629

642 # update uses of the new function, modifying call sites to include the kernel state
643 function rewrite_uses!(f, ft)
644 # update uses
645 @dispose builder=IRBuilder() begin
646 for use in uses(f)
647 val = user(use)
648 if val isa LLVM.CallBase && called_operand(val) == f
649 # NOTE: we don't rewrite calls using Julia's jlcall calling convention,
650 # as those have a fixed argument list, passing actual arguments
651 # in an array of objects. that doesn't matter, for now, since
652 # GPU back-ends don't support such calls anyhow. but if we ever
653 # want to support kernel state passing on more capable back-ends,
654 # we'll need to update the argument array instead.
655 if callconv(val) == 37 || callconv(val) == 38
656 # TODO: update for LLVM 15 when JuliaLang/julia#45088 is merged.
657 continue
658 end
659
660 # forward the state argument
661 position!(builder, val)
662 state = call!(builder, state_intr_ft, state_intr, Value[], "state")
663 new_val = if val isa LLVM.CallInst
664 call!(builder, ft, f, [state, arguments(val)...], operand_bundles(val))
665 else
666 # TODO: invoke and callbr
667 error("Rewrite of $(typeof(val))-based calls is not implemented: $val")
668 end
669 callconv!(new_val, callconv(val))
670
671 replace_uses!(val, new_val)
672 @assert isempty(uses(val))
673 erase!(val)
674 elseif val isa LLVM.CallBase
675 # the function is being passed as an argument. to avoid having to
676 # rewrite the target function, instead case the rewritten function to
677 # the old stateless type.
678 # XXX: we won't have to do this with opaque pointers.
679 position!(builder, val)
680 target_ft = called_type(val)
681 new_args = map(zip(parameters(target_ft),
682 arguments(val))) do (param_typ, arg)
683 if value_type(arg) != param_typ
684 const_bitcast(arg, param_typ)
685 else
686 arg
687 end
688 end
689 new_val = call!(builder, called_type(val), called_operand(val), new_args,
690 operand_bundles(val))
691 callconv!(new_val, callconv(val))
692
693 replace_uses!(val, new_val)
694 @assert isempty(uses(val))
695 erase!(val)
696 elseif val isa LLVM.StoreInst
697 # the function is being stored, which again we'll permit like before.
698 elseif val isa ConstantExpr
699 rewrite_uses!(val, ft)
700 else
701 error("Cannot rewrite $(typeof(val)) use of function: $val")
702 end
703 end
704 end
705 end
706 for f in values(workmap)
707 ft = function_type(f)
708 rewrite_uses!(f, ft)
709 end
710
711 return true
712 end

715 # lower calls to the state getter intrinsic. this is a two-step process, so that the state
716 # argument can be added before optimization, and that optimization can introduce new uses
717 # before the intrinsic getting lowered late during optimization.
718 function lower_kernel_state!(fun::LLVM.Function)
719 job = current_job::CompilerJob
720 mod = LLVM.parent(fun)
721 changed = false
722
723 # check if we even need a kernel state argument
724 state = kernel_state_type(job)
725 if state === Nothing
726 return false
727 end
728
729 # fixup all uses of the state getter to use the newly introduced function state argument
730 if haskey(functions(mod), "julia.gpu.state_getter")
731 state_intr = functions(mod)["julia.gpu.state_getter"]
732 state_arg = nothing # only look-up when needed
733
734 @dispose builder=IRBuilder() begin
735 for use in uses(state_intr)
736 inst = user(use)
737 @assert inst isa LLVM.CallInst
738 bb = LLVM.parent(inst)
739 LLVM.parent(bb) == fun || continue
740
741 position!(builder, inst)
742 bb = LLVM.parent(inst)
743 f = LLVM.parent(bb)
744
745 if state_arg === nothing
746 # find the kernel state argument. this should be the first argument of
747 # the function, but only when this function needs the state!
748 state_arg = parameters(fun)[1]
749 T_state = convert(LLVMType, state)
750 @assert value_type(state_arg) == T_state
751 end
752
753 replace_uses!(inst, state_arg)
754
755 @assert isempty(uses(inst))
756 erase!(inst)
757
758 changed = true
759 end
760 end
761 end
762
763 return changed
764 end

787 functions(mod)["julia.gpu.state_getter"]
788 else

796 # run-time equivalent
797 function kernel_state_value(state)
798 @dispose ctx=Context() begin
799 T_state = convert(LLVMType, state)
800
801 # create function
802 llvm_f, _ = create_function(T_state)
803 mod = LLVM.parent(llvm_f)
804
805 # get intrinsic
806 state_intr = kernel_state_intr(mod, T_state)
807 state_intr_ft = function_type(state_intr)
808
809 # generate IR
810 @dispose builder=IRBuilder() begin
811 entry = BasicBlock(llvm_f, "entry")
812 position!(builder, entry)
813
814 val = call!(builder, state_intr_ft, state_intr, Value[], "state")
815
816 ret!(builder, val)
817 end
818
819 call_function(llvm_f, state)
820 end
821 end

823 # convert kernel state argument from pass-by-value to pass-by-reference
824 #
825 # the kernel state argument is always passed by value to avoid codegen issues with byval.
826 # some back-ends however do not support passing kernel arguments by value, so this pass
827 # serves to convert that argument (and is conceptually the inverse of `lower_byval`).
828 function kernel_state_to_reference!(@nospecialize(job::CompilerJob), mod::LLVM.Module,
829 f::LLVM.Function)
830 ft = function_type(f)
831
832 # check if we even need a kernel state argument
833 state = kernel_state_type(job)
834 if state === Nothing
835 return f
836 end
837
838 T_state = convert(LLVMType, state)
839
840 # find the kernel state parameter (should be the first argument)
841 if isempty(parameters(ft)) || value_type(parameters(f)[1]) != T_state
842 return f
843 end
844
845 @tracepoint "kernel state to reference" begin
846 # generate the new function type & definition
847 new_types = LLVM.LLVMType[]
848 # convert the first parameter (kernel state) to a pointer
849 push!(new_types, LLVM.PointerType(T_state))
850 # keep all other parameters as-is
851 for i in 2:length(parameters(ft))
852 push!(new_types, parameters(ft)[i])
853 end
854
855 new_ft = LLVM.FunctionType(return_type(ft), new_types)
856 new_f = LLVM.Function(mod, "", new_ft)
857 linkage!(new_f, linkage(f))
858
859 # name the parameters
860 LLVM.name!(parameters(new_f)[1], "state_ptr")
861 for (i, (arg, new_arg)) in enumerate(zip(parameters(f)[2:end], parameters(new_f)[2:end]))
862 LLVM.name!(new_arg, LLVM.name(arg))
863 end
864
865 # emit IR performing the "conversions"
866 new_args = LLVM.Value[]
867 @dispose builder=IRBuilder() begin
868 entry = BasicBlock(new_f, "conversion")
869 position!(builder, entry)
870
871 # load the kernel state value from the pointer
872 state_val = load!(builder, T_state, parameters(new_f)[1], "state")
873 push!(new_args, state_val)
874
875 # all other arguments are passed through directly
876 for i in 2:length(parameters(new_f))
877 push!(new_args, parameters(new_f)[i])
878 end
879
880 # map the arguments
881 value_map = Dict{LLVM.Value, LLVM.Value}(
882 param => new_args[i] for (i,param) in enumerate(parameters(f))
883 )
884 value_map[f] = new_f
885
886 clone_into!(new_f, f; value_map,
887 changes=LLVM.API.LLVMCloneFunctionChangeTypeGlobalChanges)
888
889 # fall through
890 br!(builder, blocks(new_f)[2])
891 end
892
893 # set the attributes for the state pointer parameter
894 attrs = parameter_attributes(new_f, 1)
895 # the pointer itself cannot be captured since we immediately load from it
896 push!(attrs, EnumAttribute("nocapture", 0))
897 # each kernel state is separate
898 push!(attrs, EnumAttribute("noalias", 0))
899 # the state is read-only
900 push!(attrs, EnumAttribute("readonly", 0))
901
902 # remove the old function
903 fn = LLVM.name(f)
904 @assert isempty(uses(f))
905 replace_metadata_uses!(f, new_f)
906 erase!(f)
907 LLVM.name!(new_f, fn)
908
909 # minimal optimization
910 @dispose pb=NewPMPassBuilder() begin
911 add!(pb, SimplifyCFGPass())
912 run!(pb, new_f, llvm_machine(job.config.target))
913 end
914
915 return new_f
916 end
917 end
918
919 function add_input_arguments!(@nospecialize(job::CompilerJob), mod::LLVM.Module,
920 entry::LLVM.Function, kernel_intrinsics::Dict)
921 entry_fn = LLVM.name(entry)
922
923 # figure out which intrinsics are used and need to be added as arguments
924 used_intrinsics = filter(keys(kernel_intrinsics)) do intr_fn
925 haskey(functions(mod), intr_fn)
926 end |> collect
927 nargs = length(used_intrinsics)
928
929 # determine which functions need these arguments
930 worklist = Set{LLVM.Function}([entry])
931 for intr_fn in used_intrinsics
932 push!(worklist, functions(mod)[intr_fn])
933 end
934 worklist_length = 0
935 while worklist_length != length(worklist)
936 # iteratively discover functions that use an intrinsic or any function calling it
937 worklist_length = length(worklist)
938 additions = Set{LLVM.Function}()
939 function scan_uses(val)
940 for use in uses(val)
941 candidate = user(use)
942 if isa(candidate, Instruction)
943 bb = LLVM.parent(candidate)
944 new_f = LLVM.parent(bb)
945 in(new_f, worklist) || push!(additions, new_f)
946 elseif isa(candidate, ConstantExpr)
947 scan_uses(candidate)
948 else
949 error("Don't know how to check uses of $candidate. Please file an issue.")
950 end
951 end
952 end
953 for f in worklist
954 scan_uses(f)
955 end
956 for f in additions
957 push!(worklist, f)
958 end
959 end
960 for intr_fn in used_intrinsics
961 delete!(worklist, functions(mod)[intr_fn])
962 end
963
964 # add the arguments
965 # NOTE: we don't need to be fine-grained here, as unused args will be removed during opt
966 workmap = Dict{LLVM.Function, LLVM.Function}()
967 for f in worklist
968 fn = LLVM.name(f)
969 ft = function_type(f)
970 LLVM.name!(f, fn * ".orig")
971 # create a new function
972 new_param_types = LLVMType[parameters(ft)...]
973
974 for intr_fn in used_intrinsics
975 llvm_typ = convert(LLVMType, kernel_intrinsics[intr_fn].typ)
976 push!(new_param_types, llvm_typ)
977 end
978 new_ft = LLVM.FunctionType(return_type(ft), new_param_types)
979 new_f = LLVM.Function(mod, fn, new_ft)
980 linkage!(new_f, linkage(f))
981 for (arg, new_arg) in zip(parameters(f), parameters(new_f))
982 LLVM.name!(new_arg, LLVM.name(arg))
983 end
984 for (intr_fn, new_arg) in zip(used_intrinsics, parameters(new_f)[end-nargs+1:end])
985 LLVM.name!(new_arg, kernel_intrinsics[intr_fn].name)
986 end
987
988 workmap[f] = new_f
989 end
990
991 # clone and rewrite the function bodies.
992 # we don't need to rewrite much as the arguments are added last.
993 for (f, new_f) in workmap
994 # map the arguments
995 value_map = Dict{LLVM.Value, LLVM.Value}()
996 for (param, new_param) in zip(parameters(f), parameters(new_f))
997 LLVM.name!(new_param, LLVM.name(param))
998 value_map[param] = new_param
999 end
1000
1001 value_map[f] = new_f
1002 clone_into!(new_f, f; value_map,
1003 changes=LLVM.API.LLVMCloneFunctionChangeTypeLocalChangesOnly)
1004
1005 # we can't remove this function yet, as we might still need to rewrite any called,
1006 # but remove the IR already
1007 empty!(f)
1008 end
1009
1010 # drop unused constants that may be referring to the old functions
1011 # XXX: can we do this differently?
1012 for f in worklist
1013 prune_constexpr_uses!(f)
1014 end
1015
1016 # update other uses of the old function, modifying call sites to pass the arguments
1017 function rewrite_uses!(f, new_f)
1018 # update uses
1019 @dispose builder=IRBuilder() begin
1020 for use in uses(f)
1021 val = user(use)
1022 if val isa LLVM.CallInst || val isa LLVM.InvokeInst || val isa LLVM.CallBrInst
1023 callee_f = LLVM.parent(LLVM.parent(val))
1024 # forward the arguments
1025 position!(builder, val)
1026 new_val = if val isa LLVM.CallInst
1027 call!(builder, function_type(new_f), new_f,
1028 [arguments(val)..., parameters(callee_f)[end-nargs+1:end]...],
1029 operand_bundles(val))
1030 else
1031 # TODO: invoke and callbr
1032 error("Rewrite of $(typeof(val))-based calls is not implemented: $val")
1033 end
1034 callconv!(new_val, callconv(val))
1035
1036 replace_uses!(val, new_val)
1037 @assert isempty(uses(val))
1038 erase!(val)
1039 elseif val isa LLVM.ConstantExpr && opcode(val) == LLVM.API.LLVMBitCast
1040 # XXX: why isn't this caught by the value materializer above?
1041 target = operands(val)[1]
1042 @assert target == f
1043 new_val = LLVM.const_bitcast(new_f, value_type(val))
1044 rewrite_uses!(val, new_val)
1045 # we can't simply replace this constant expression, as it may be used
1046 # as a call, taking arguments (so we need to rewrite it to pass the input arguments)
1047
1048 # drop the old constant if it is unused
1049 # XXX: can we do this differently?
1050 if isempty(uses(val))
1051 LLVM.unsafe_destroy!(val)
1052 end
1053 else
1054 error("Cannot rewrite unknown use of function: $val")
1055 end
1056 end
1057 end
1058 end
1059 for (f, new_f) in workmap
1060 rewrite_uses!(f, new_f)
1061 @assert isempty(uses(f))
1062 replace_metadata_uses!(f, new_f)
1063 erase!(f)
1064 end
1065
1066 # replace uses of the intrinsics with references to the input arguments
1067 for (i, intr_fn) in enumerate(used_intrinsics)
1068 intr = functions(mod)[intr_fn]
1069 for use in uses(intr)
1070 val = user(use)
1071 callee_f = LLVM.parent(LLVM.parent(val))
1072 if val isa LLVM.CallInst || val isa LLVM.InvokeInst || val isa LLVM.CallBrInst
1073 replace_uses!(val, parameters(callee_f)[end-nargs+i])
1074 else
1075 error("Cannot rewrite unknown use of function: $val")
1076 end
1077
1078 @assert isempty(uses(val))
1079 erase!(val)
1080 end
1081 @assert isempty(uses(intr))
1082 erase!(intr)
1083 end
1084
1085 return functions(mod)[entry_fn]
1086 end
