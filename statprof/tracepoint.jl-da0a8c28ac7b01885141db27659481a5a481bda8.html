6 function _extract_color(ex)
7 kws, body = extract_keywords(ex)
8 for (k, v) in kws
9 k === :color || continue
10 color = v
11 if color !== 0
12 if color isa Integer
13 color = _tracycolor(color)
14 elseif color isa QuoteNode
15 color = _tracycolor(color.value)
16 elseif Meta.isexpr(color, :tuple)
17 if length(color.args) == 3 && all(isa.(color.args, Integer))
18 color = _tracycolor(Tuple(color.args))
19 else
20 error("Invalid color tuple: $color")
21 end
22 else
23 error("Invalid color: $color")
24 end
25 end
26 kws[k] = color
27 end
28 return body, kws
29 end

35 """
36 @tracepoint "name" color=<color> enabled=<expr> <expression>
37
38 Code you'd like to trace should be wrapped with `@tracepoint`
39
40 ```julia
41 @tracepoint "name" <expression>
42 ```
43
44 Typically the expression will be a `begin-end` block:
45
46 ```julia
47 @tracepoint "data aggregation" begin
48 # lots of compute here...
49 end
50 ```
51
52 The name of the tracepoint must be a literal string, and it cannot
53 be changed at runtime. The tracepoint can be dynamically disabled or enabled
54 by using the `enabled` keyword argument which should be a boolean expression.
55
56 The (default) color of the zone can be configured
57 with the `color` keyword argument to the macro which should be a literal that can either be:
58 $color_docstr
59
60 You can also trace function definitions where name of the tracepoint
61 will be the name of the function unless it is explicitly provided:
62
63 ```julia
64 @tracepoint function f(x)
65 x^2
66 end
67
68 @tracepoint "calling g" g(x) = x^2
69
70 h = @tracepoint x -> x^2
71 ```
72
73 If you don't have Tracy installed, you can install `TracyProfiler_jll`
74 and start it with `run(TracyProfiler_jll.tracy(); wait=false)`.
75
76 ```jldoctest
77 julia> x = rand(10,10);
78
79 julia> @tracepoint "multiply" x * x;
80
81 julia> @tracepoint "pow2" color=:green x^2; # green
82
83 julia> @tracepoint "pow3" color=:0xFF0000 x^3; # red
84
85 julia> @tracepoint "pow4" color=(255, 165, 0) x^4; # orange
86
87 julia> timings_enabled() = rand() < 0.5;
88
89 julia> @tracepoint "pow5" enabled=timings_enabled() x^5; # enabled only if timings_enabled() is true
90 ```
91
92 If you don't have Tracy installed, you can install `TracyProfiler_jll`
93 and start it with `run(TracyProfiler_jll.tracy(); wait=false)`.
94 """

206 """
207 configure_tracepoint
208
209 Enable/disable a set of tracepoint(s) in the provided modules by invalidating any
210 existing code containing the tracepoint(s).
211
212 !!! warning
213 This invalidates the code generated for all functions containing the selected zones.
214
215 This will trigger re-compilation for these functions and may cause undesirable latency.
216 It is strongly recommended to use `enable_tracepoint` instead.
217 """
218 function configure_tracepoint(m::Module, enable::Bool; name="", func="", file="")
219 m_id = invokelatest(getfield, m, ID)
220 for (i, srcloc) in enumerate(tracymeta(m))
221 contains(srcloc.name, name) || continue
222 contains(srcloc.func, func) || continue
223 contains(srcloc.file, file) || continue
224 Core.eval(m, :($Tracy.tracepoint_enabled(::Val{$m_id}, ::Val{$i}) = $enable))
225 end
226 return nothing
227 end

229 """
230 enable_tracepoint
231
232 Enable/disable a set of tracepoint(s) in the provided modules, based on whether they
233 match the filters provided for `name`/`func`/`file`.
234 """
235 function enable_tracepoint(m::Module, enable::Bool; name="", func="", file="")
236 for srcloc in tracymeta(m)
237 contains(srcloc.name, name) || continue
238 contains(srcloc.func, func) || continue
239 contains(srcloc.file, file) || continue
240 srcloc.enabled = enable
241 end
242 return nothing
243 end
