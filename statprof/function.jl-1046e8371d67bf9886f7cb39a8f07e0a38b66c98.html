291 # provide a random access interface by maintaining a cache of blocks
292 function Base.getindex(iter::FunctionBlockSet, i::Int)
293 i <= 0 && throw(BoundsError(iter, i))
294 i == 1 && return first(iter)
295 while i > length(iter.cache)
296 next = if isempty(iter.cache)
297 iterate(iter)
298 else
299 iterate(iter, iter.cache[end])
300 end
301 next === nothing && throw(BoundsError(iter, i))
302 push!(iter.cache, next[2])
303 end
304 return BasicBlock(iter.cache[i-1])
305 end
