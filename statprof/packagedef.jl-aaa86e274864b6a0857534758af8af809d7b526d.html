268 ##
269 ## The inputs are sets of expressions found in each file.
270 ## Some of those expressions will generate methods which are identified via their signatures.
271 ## From "old" expressions we know their corresponding signatures, but from "new"
272 ## expressions we have not yet computed them. This makes old and new asymmetric.
273 ##
274 ## Strategy:
275 ## - For every old expr not found in the new ones,
276 ## + delete the corresponding methods (using the signatures we've previously computed)
277 ## + remove the sig entries from CodeTracking.method_info (")
278 ## Best to do all the deletion first (across all files and modules) in case a method is
279 ## simply being moved from one file to another.
280 ## - For every new expr found among the old ones,
281 ## + update the location info in CodeTracking.method_info
282 ## - For every new expr not found in the old ones,
283 ## + eval the expr
284 ## + extract signatures
285 ## + add to the ModuleExprsSigs
286 ## + add to CodeTracking.method_info
287 ##
288 ## Interestingly, the ex=>mt_sigs link may not be the same as the mt_sigs=>ex link.
289 ## Consider a conditional block,
290 ## if Sys.islinux()
291 ## f() = 1
292 ## g() = 2
293 ## else
294 ## g() = 3
295 ## end
296 ## From the standpoint of Revise's diff-and-patch functionality, we should look for
297 ## diffs in this entire block. (Really good backedge support---or a variant of `lower` that
298 ## links back to the specific expression---might change this, but for
299 ## now this is the right strategy.) From the standpoint of CodeTracking, we should
300 ## link the signature to the actual method-defining expression (either :(f() = 1) or :(g() = 2)).

644 # issue #459
645 (isa(e, InterruptException) && throwto_repl(e)) || throw(e)
646 end

648 if file in keys(user_callbacks_by_file)
649 union!(user_callbacks_queue, user_callbacks_by_file[file])
650 notify(revision_event)
651 end
652
653 # Check to see if we're still watching this file
654 stillwatching = haskey(watched_files, dirfull)
655 if PkgId(pkgdata) != NOPACKAGE
656 lock(revision_queue_lock) do
657 push!(revision_queue, (pkgdata, relpath(file, pkgdata)))
658 end
659 end
660 end
661 return

762 """
763 revise(; throw=false)
764
765 `eval` any changes in the revision queue. See [`Revise.revision_queue`](@ref).
766 If `throw` is `true`, throw any errors that occur during revision or callback;
767 otherwise these are only logged.
768 """
769 function revise(; throw::Bool=false)
770 active[] || return nothing
771 sleep(0.01) # in case the file system isn't quite done writing out the new files
772 lock(revision_queue_lock) do
773 have_queue_errors = !isempty(queue_errors)
774
775 # Do all the deletion first. This ensures that a method that moved from one file to another
776 # won't get redefined first and deleted second.
777 revision_errors = Tuple{PkgData,String}[]
778 queue = sort!(collect(revision_queue); lt=pkgfileless)
779 finished = eltype(revision_queue)[]
780 mexsnews = ModuleExprsSigs[]
781 interrupt = false
782 for (pkgdata, file) in queue
783 try
784 mexsnew, _ = handle_deletions(pkgdata, file)
785 mexsnew === DoNotParse() && continue
786 push!(mexsnews, mexsnew)
787 push!(finished, (pkgdata, file))
788 catch err
789 throw && Base.throw(err)
790 interrupt |= isa(err, InterruptException)
791 push!(revision_errors, (pkgdata, file))
792 queue_errors[(pkgdata, file)] = (err, catch_backtrace())
793 end
794 end
795 # Do the evaluation
796 for ((pkgdata, file), mexsnew) in zip(finished, mexsnews)
797 defaultmode = PkgId(pkgdata).name == "Main" ? :evalmeth : :eval
798 i = fileindex(pkgdata, file)
799 i === nothing && continue # file was deleted by `handle_deletions`
800 fi = fileinfo(pkgdata, i)
801 modsremaining = Set(keys(mexsnew))
802 changed, err = true, nothing
803 while changed
804 changed = false
805 for (mod, exsnew) in mexsnew
806 mod ∈ modsremaining || continue
807 try
808 mode = defaultmode
809 # Allow packages to override the supplied mode
810 if isdefined(mod, :__revise_mode__)
811 mode = getfield(mod, :__revise_mode__)::Symbol
812 end
813 mode ∈ (:sigs, :eval, :evalmeth, :evalassign) || error("unsupported mode ", mode)
814 exsold = get(fi.modexsigs, mod, empty_exs_sigs)
815 for rex in keys(exsnew)
816 mt_sigs, includes = eval_rex(rex, exsold, mod; mode=mode)
817 if mt_sigs !== nothing
818 exsnew[rex] = mt_sigs
819 end
820 if includes !== nothing
821 maybe_add_includes_to_pkgdata!(pkgdata, file, includes; eval_now=true)
822 end
823 end
824 delete!(modsremaining, mod)
825 changed = true
826 catch _err
827 err = _err
828 end
829 end
830 end
831 if isempty(modsremaining) || isa(err, LoweringException) # fix #877
832 pkgdata.fileinfos[i] = FileInfo(mexsnew, fi)
833 end
834 if isempty(modsremaining)
835 delete!(queue_errors, (pkgdata, file))
836 else
837 throw && Base.throw(err)
838 interrupt |= isa(err, InterruptException)
839 push!(revision_errors, (pkgdata, file))
840 queue_errors[(pkgdata, file)] = (err, catch_backtrace())
841 end
842 end
843 if interrupt
844 for pkgfile in finished
845 haskey(queue_errors, pkgfile) || delete!(revision_queue, pkgfile)
846 end
847 else
848 empty!(revision_queue)
849 end
850 errors(revision_errors)
851 if !isempty(queue_errors)
852 if !have_queue_errors # only print on the first time errors occur
853 io = IOBuffer()
854 println(io, "\n") # better here than in the triple-quoted literal, see https://github.com/JuliaLang/julia/issues/34105
855 for (pkgdata, file) in keys(queue_errors)
856 println(io, " ", joinpath(basedir(pkgdata), file))
857 end
858 str = String(take!(io))
859 @warn """The running code does not match the saved version for the following files:$str
860 If the error was due to evaluation order, it can sometimes be resolved by calling `Revise.retry()`.
861 Use Revise.errors() to report errors again. Only the first error in each file is shown.
862 Your prompt color may be yellow until the errors are resolved."""
863 maybe_set_prompt_color(:warn)
864 end
865 else
866 maybe_set_prompt_color(:ok)
867 end
868 tracking_Main_includes[] && queue_includes(Main)
869
870 process_user_callbacks!(; throw)
871 end
872
873 nothing
874 end

964 push!(pkgdata, relpath(file, pkgdata)=>FileInfo(fm))

967 end
968 end
969 return nothing

977 """
978 includet(filename::AbstractString)
979
980 Load `filename` and track future changes. `includet` is intended for quick "user scripts"; larger or more
981 established projects are encouraged to put the code in one or more packages loaded with `using`
982 or `import` instead of using `includet`. See https://timholy.github.io/Revise.jl/stable/cookbook/
983 for tips about setting up the package workflow.
984
985 By default, `includet` only tracks modifications to *methods*, not *data*. See the extended help for details.
986 Note that this differs from packages, which evaluate all changes by default.
987 This default behavior can be overridden; see [Configuring the revise mode](@ref).
988
989 # Extended help
990
991 ## Behavior and justification for the default revision mode (`:evalmeth`)
992
993 `includet` uses a default `__revise_mode__ = :evalmeth`. The consequence is that if you change
994
995 ```
996 a = [1]
997 f() = 1
998 ```
999 to
1000 ```
1001 a = [2]
1002 f() = 2
1003 ```
1004 then Revise will update `f` but not `a`.
1005
1006 This is the default choice for `includet` because such files typically mix method definitions and data-handling.
1007 Data often has many untracked dependencies; later in the same file you might `push!(a, 22)`, but Revise cannot
1008 determine whether you wish it to re-run that line after redefining `a`.
1009 Consequently, the safest default choice is to leave the user in charge of data.
1010
1011 ## Workflow tips
1012
1013 If you have a series of computations that you want to run when you redefine your methods, consider separating
1014 your method definitions from your computations:
1015
1016 - method definitions go in a package, or a file that you `includet` *once*
1017 - the computations go in a separate file, that you re-`include` (no "t" at the end) each time you want to rerun
1018 your computations.
1019
1020 This can be automated using [`entr`](@ref).
1021
1022 ## Internals
1023
1024 `includet` is essentially shorthand for
1025
1026 Revise.track(Main, filename; mode=:includet, skip_include=true)
1027
1028 Do *not* use `includet` for packages, as those should be handled by `using` or `import`.
1029 If `using` and `import` aren't working, you may have packages in a non-standard location;
1030 try fixing it with something like `push!(LOAD_PATH, "/path/to/my/private/repos")`.
1031 (If you're working with code in Base or one of Julia's standard libraries, use
1032 `Revise.track(mod)` instead, where `mod` is the module.)
1033
1034 `includet` is deliberately non-recursive, so if `filename` loads any other files,
1035 they will not be automatically tracked.
1036 (Call [`Revise.track`](@ref) manually on each file, if you've already `included`d all the code you need.)
1037 """
1038 function includet(mod::Module, file::AbstractString)
1039 prev = Base.source_path(nothing)
1040 file = if prev === nothing
1041 abspath(file)
1042 else
1043 normpath(joinpath(dirname(prev), file))
1044 end
1045 tls = task_local_storage()
1046 tls[:SOURCE_PATH] = file
1047 try
1048 track(mod, file; mode=:includet, skip_include=true)
1049 if prev === nothing
1050 delete!(tls, :SOURCE_PATH)
1051 else
1052 tls[:SOURCE_PATH] = prev
1053 end
1054 catch err
1055 if prev === nothing
1056 delete!(tls, :SOURCE_PATH)
1057 else
1058 tls[:SOURCE_PATH] = prev
1059 end
1060 if isa(err, ReviseEvalException)
1061 printstyled(stderr, "ERROR: "; color=Base.error_color());
1062 invokelatest(showerror, stderr, err; blame_revise=false)
1063 println(stderr, "\nin expression starting at ", err.loc)
1064 else
1065 throw(err)
1066 end
1067 end
1068 return nothing
1069 end
1070 includet(file::AbstractString) = includet(Main, file)

1103 yield() # magic bug fix for the OSX test failures. TODO: figure out why this works (prob. Julia bug)
1104 if method.file === :none && String(method.name)[1] == '#'

1268 # `revise_first` gets called by the REPL prior to executing the next command (by having been pushed
1269 # onto the `ast_transform` list).
1270 # This uses invokelatest not for reasons of world age but to ensure that the call is made at runtime.
1271 # This allows `revise_first` to be compiled without compiling `revise` itself, and greatly
1272 # reduces the overhead of using Revise.
1273 function revise_first(ex)
1274 # Special-case `exit()` (issue #562)
1275 if isa(ex, Expr)
1276 exu = unwrap(ex)
1277 if isexpr(exu, :block, 2)
1278 arg1 = exu.args[1]
1279 if isexpr(arg1, :softscope)
1280 exu = exu.args[2]
1281 end
1282 end
1283 if isa(exu, Expr)
1284 exu.head === :call && length(exu.args) == 1 && exu.args[1] === :exit && return ex
1285 lhsrhs = LoweredCodeUtils.get_lhs_rhs(exu)
1286 if lhsrhs !== nothing

1434 function watch_package_callback(id::PkgId)
1435 # `Base.package_callbacks` fire immediately after module initialization, and
1436 # would fire on Revise itself. This is not necessary for most users, and has
1437 # the downside that the user doesn't get to the REPL prompt until
1438 # `watch_package` finishes compiling. To prevent this, Revise hides the
1439 # actual `watch_package` method behind an `invokelatest`. This delays
1440 # compilation of everything that `watch_package` requires, leading to faster
1441 # perceived startup times.
1442 if id != REVISE_ID
1443 Base.invokelatest(watch_package, id)
1444 end
1445 return
1446 end
